# ⚠️ 필수 사전 설치 (Windows 터미널에서 실행, pip install과 별개):
# winget install ffmpeg
# 설치 후 터미널/VS Code를 완전히 재시작해야 PATH가 반영됩니다.

fastapi                         # 루나 프록시 서버의 웹 프레임워크 (API 엔드포인트 정의)
uvicorn[standard]               # FastAPI 앱을 실행하는 ASGI 서버 (standard=성능 최적화 의존성 포함)
openai                          # Azure AI Foundry(gpt-5.6-luna) 호출용 클라이언트 SDK
python-dotenv                   # .env 파일에서 환경변수(API 키, 엔드포인트 등) 로드
azure-cognitiveservices-speech  # 페르소나별 TTS 음성 합성(무브/토닥이/척척박사/링고) 및 음성 인식(STT) 처리
psycopg2-binary                 # Azure PostgreSQL Flexible Server(moov-postgres-eastus2) 연결용 DB 드라이버
pgvector                        # PostgreSQL의 pgvector 확장을 Python에서 다루기 위한 라이브러리 (벡터 RAG 검색용)
pydub                           # STT용 오디오 포맷 변환(webm→wav) — Azure Speech가 요구하는 WAV(16kHz mono)로 맞추는 데 사용, ffmpeg 필요
audioop-lts                     # Python 3.13에서 pydub가 필요로 하는 audioop 대체 (3.13이 표준 audioop 모듈을 제거함)
python-multipart                # FastAPI가 UploadFile(오디오 파일)을 받으려면 필요
azure-storage-blob              # 코스 등록 사진을 Azure Blob Storage(course-images 컨테이너)에 업로드하기 위한 라이브러리
